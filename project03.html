// 003 — инструмент / опен сурс

NOISE
CLI

генератор шумовых ASCII-текстур для терминала.
чистый python 3, без зависимостей.

python 3.7+ zero deps MIT

// установка

# запустить напрямую
$ python noise_cli.py gen perlin --color
# установить глобально
$ pip install .
$ noise gen perlin --color

// использование

noise gen [type] [options]
# типы шума
type perlin | smooth | fractal | cellular | warp | white
# опции
-W --width N ширина в символах (default: 80)
-H --height N высота в строках (default: 40)
-s --scale F масштаб шума (default: 4.0)
-p --palette NAME символьная палитра (default: void)
--seed N зерно генератора (default: random)
--color ANSI-цвет (зелёный фосфор)
--animate цикличная анимация
--fps N кадров/сек (default: 10)
-o --output FILE сохранить в .txt

// примеры

$ noise gen perlin --color

базовый FBM, 4 октавы

$ noise gen fractal --palette dots --seed 1337 --color

детализированный FBM, 6 октавов

$ noise gen cellular --palette matrix --color

Worley noise — органические ячейки

$ noise gen warp --scale 6 --color

domain warping — закрученные формы

$ noise gen fractal --animate --fps 15 --color

живая анимация шума

$ noise gen perlin -o texture.txt

сохранить в файл
